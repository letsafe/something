local function FindTopBarApp()

    if typeof(Project) ==
        "Instance"
    then

        if Project:IsA("ScreenGui")
            and Project.Name ==
                "TopBarApp"
        then

            return Project

        end

        local Found =
            Project:FindFirstChild(
                "TopBarApp",
                true
            )

        if Found then
            return Found
        end
    end

    return CoreGui:FindFirstChild(
        "TopBarApp",
        true
    )
end

        { ... }

-- SAFE PROPERTY HELPERS

local function SafeRead(
    Object,
    Property
)

    if not Object then
        return false, nil
    end

    return pcall(function()
        return Object[Property]
    end)
end

local function CopyImageState(
    Source,
    Target
)

    if not Source
        or not Target
    then

        return

    end

    for _, Property in ipairs({

        "Image",
        "ImageColor3",
        "ImageTransparency",
        "ImageRectOffset",
        "ImageRectSize"

    }) do

        local Success,
            Value =
            SafeRead(
                Source,
                Property
            )

        if Success then

            pcall(function()

                Target[Property] =
                    Value

            end)

        end
    end
end

--============================================================
-- DROPDOWN TEXT WIDTH FIX
--============================================================

local function FixDropdownLabel(
    Label,
    Row
)

    if not Label then
        return
    end

    pcall(function()
        Label.Text = "Dropdown"
        Label.TextWrapped = false
        Label.ClipsDescendants = false
        Label.TextTruncate = Enum.TextTruncate.None
        Label.AutomaticSize = Enum.AutomaticSize.X
        Label.Visible = true
    end)

    -- Only the label's immediate text host and row may allow overflow.
    local Parent = Label.Parent
    if Parent then
        pcall(function()
            if Parent:IsA("GuiObject") then
                Parent.ClipsDescendants = false
            end
        end)
    end

    if Row then
        pcall(function()
            Row.ClipsDescendants = false
        end)
    end

    -- The submenu containers themselves must always clip.
    pcall(function()
        if SubMenuHost then SubMenuHost.ClipsDescendants = false end
        if NineDotMenu then NineDotMenu.ClipsDescendants = true end
        if ScrollingFrame then ScrollingFrame.ClipsDescendants = true end
    end)

    task.defer(function()
        if not Label or not Label.Parent then
            return
        end

        pcall(function()
            local Bounds = Label.TextBounds.X
            local OldSize = Label.Size
            local Width = math.max(78, math.ceil(Bounds + 8))

            Label.Size = UDim2.new(
                0,
                Width,
                OldSize.Y.Scale,
                OldSize.Y.Offset
            )
        end)
    end)
end

local function CreateExtraSubmenuRow(
    Source,
    Name,
    Text
)

    if not Source
        or not ScrollingFrame
    then
        return nil
    end

    local Existing =
        ScrollingFrame:FindFirstChild(
            Name
        )

    if Existing then
        return Existing
    end

    local Row =
        Source:Clone()

    Row.Name =
        Name

    Row.Parent =
        ScrollingFrame

    local Label =
        Row:FindFirstChild(
            "StyledTextLabel",
            true
        )

    if Label then

        pcall(function()
            Label.Text =
                Text
        end)

        if Name == "nine_dot_submenu" then
            FixDropdownLabel(Label, Row)
        end

    end

    return Row
end

local TransparencyProperties = {

    "BackgroundTransparency",
    "ImageTransparency",
    "TextTransparency",
    "TextStrokeTransparency",
    "ScrollBarImageTransparency",
    "GroupTransparency",
    "Transparency"

}

local TransparencyCache = {}

local function CacheTransparency(
    Object
)

    if not Object then
        return
    end

    -- PinButton/PinGlyph are controls, not submenu fade content.
    -- Never cache them because the submenu fade would otherwise
    -- set their ImageTransparency to 1 and make the pin disappear.
    if Object.Name == "PinButton"
        or Object.Name == "PinGlyph"
    then
        return
    end

    -- Also exclude every descendant of a PinButton.
    local Parent = Object.Parent

    while Parent do

        if Parent.Name == "PinButton" then
            return
        end

        Parent = Parent.Parent
    end

    local Cache =
        {}

    for _, Property in ipairs(
        TransparencyProperties
    ) do

        local Success,
            Value =
            SafeRead(
                Object,
                Property
            )

        if Success
            and type(Value) ==
                "number"
        then

            Cache[Property] =
                Value

        end
    end

    TransparencyCache[
        Object
    ] =
        Cache
end

CacheTransparency(
    SubMenuHost
)

if SubMenuHost then

    for _, Object in ipairs(
        SubMenuHost:GetDescendants()
    ) do

        CacheTransparency(
            Object
        )

    end
end

--============================================================
-- NATIVE ACTION LOCKS
--============================================================
-- A click updates the custom icon immediately. The native CoreGui
-- can take a few frames to catch up, so polling is temporarily
-- prevented from replacing the just-requested state with stale data.
--============================================================

local ActionStateLockUntil = {

    chat = 0,
    leaderboard = 0,
    backpack = 0,
    emotes = 0

}

-- Native keyboard toggles can update CoreScripts a few frames after
-- UserInputService sees the key. Keep a short predicted state so a
-- stale GetCore value cannot immediately undo the visual update.
local NativeHotkeyOverrideUntil = {

    leaderboard = 0,
    backpack = 0
}

-- A synthetic Backquote sent by our custom Backpack button can also
-- arrive at this script's own InputBegan handler. Suppress that one
-- synthetic event so a click cannot toggle Backpack twice.
local SuppressSyntheticBackquoteUntil = 0
local SuppressSyntheticTabUntil = 0

local function SetActionLocalState(
    ActionName,
    Open,
    LockTime
)

    if ActionName == "chat" then

        ChatOpen = Open

    elseif ActionName == "leaderboard" then

        LeaderboardOpen = Open

    elseif ActionName == "backpack" then

        InventoryOpen = Open

    elseif ActionName == "emotes" then

        EmotesOpen = Open

    end

    ActionStateLockUntil[ActionName] =
        os.clock() + (LockTime or 0.35)
end

local function IsActionLocked(
    ActionName
)

    return os.clock() <
        (ActionStateLockUntil[ActionName] or 0)
end

        SubmenuTextObjects
    )

    if not ScrollingFrame then
        return
    end

    for _, Row in ipairs(
        ScrollingFrame:GetChildren()
    ) do

        local Name =
            string.lower(Row.Name)

            SubmenuText[Name]

local function FadeSubmenuClosed()

    -- Text is hidden instantly. Never tween TextTransparency.
    for Label in pairs(
        SubmenuTextObjects
    ) do

        if Label
            and Label.Parent
        then

            pcall(function()
                Label.Visible = true
                Label.TextTransparency = 1
            end)
        end
    end

    -- Tween only non-text submenu artwork/background properties.
    for Object, Cache in pairs(
        TransparencyCache
    ) do

        if Object
            and Object.Parent
            and not (
                Object:IsA("TextLabel")
                or Object:IsA("TextButton")
                or Object:IsA("TextBox")
            )
        then

            local Goal =
                {}

            for Property in pairs(
                Cache
            ) do

                Goal[Property] =
                    1

            end

            if next(Goal) then

                Tween(
                    Object,
                    SubMenuTweenInfo,
                    Goal
                )

            end
        end
    end

    if type(_G.__OldChrome2024RestorePinGlyphs) == "function" then
        _G.__OldChrome2024RestorePinGlyphs()
    end
end

local function OpenSubmenu()

    if SubMenuOpen then
        return
    end

    SubMenuOpen =
        true

    if SubMenuHost then

        pcall(function()
            SubMenuHost.Visible = true
            -- The host contains the popup, so it must not clip the popup.
            SubMenuHost.ClipsDescendants = false
        end)

    end

    if NineDotMenu then
        pcall(function()
            NineDotMenu.ClipsDescendants = true
            NineDotMenu.Visible = true
        end)
    end

    if ScrollingFrame then
        pcall(function()
            ScrollingFrame.ClipsDescendants = true
            ScrollingFrame.Visible = true
        end)
    end

    CacheSubmenuText()
    SyncSubmenuIcons()
    FadeSubmenuOpen()
    ForceSubmenuTextVisible()

    ShowHighlight(
        NineDotSelected,
        0.9
    )
end

    if not SubMenuOpen then
        return
    end

    SubMenuOpen =
        false

        if NineDotMenu then NineDotMenu.ClipsDescendants = true end
        if ScrollingFrame then ScrollingFrame.ClipsDescendants = true end
    end)

    FadeSubmenuClosed()

    if SubMenuHost then
        pcall(function()
            SubMenuHost.Visible = false
        end)
    end

    RestoreAllPinGlyphs()
end

--============================================================
-- NATIVE ACTION BUTTON FINDER (ONLY 3 ACTIONS)
--============================================================
--
-- Used only when one of the three custom action buttons is clicked.
-- This is intentionally NOT a watcher/poller and does not touch the
-- rest of the Chrome UI.
--============================================================

local NativeActionButtonCache = {}

local NativeActionButtonNames = {
    leaderboard = {
        "playerlistbutton",
        "playerlist",
        "leaderboardbutton",
        "leaderboard",
        "playersbutton"
    },
    backpack = {
        "backpackbutton",
        "backpack",
        "inventorybutton",
        "inventory"
    },
    emotes = {
        "emotesbutton",
        "emotesmenubutton",
        "emotesmenu",
        "emotes"
    }
}

local function FindNativeActionButton(
    ActionName
)

    local Cached =
        NativeActionButtonCache[ActionName]

    if Cached
        and Cached.Parent
        and Cached:IsA("GuiButton")
    then

        return Cached

    end

    local Names =
        NativeActionButtonNames[ActionName]

    if not Names then
        return nil
    end

    local Best =
        nil

    local BestScore =
        -math.huge

    local NameScores =
        {}

    for Index, Name in ipairs(Names) do
        NameScores[Name] =
            1000 - Index * 20
    end

    for _, Object in ipairs(
        CoreGui:GetDescendants()
    ) do

        if not Object:IsA("GuiButton") then
            continue
        end

        if Object:GetAttribute(
            "__OldChrome2024"
        ) == true
        then

            continue
        end

        -- Never allow the native-action finder to select one of our
        -- replacement buttons. The custom TopBarApp is also a CoreGui
        -- descendant, so checking only the attribute on its root is not
        -- enough; descendants do not inherit that attribute.
        local IsOurButton =
            false

        pcall(function()
            IsOurButton =
                Object:IsDescendantOf(
                    TopBarApp
                )
        end)

        if IsOurButton then
            continue
        end

        local ObjectName =
            string.lower(
                Object.Name
            )

        local Score =
            NameScores[ObjectName]
            or 0

        local ObjectText =
            ""

        pcall(function()
            ObjectText =
                string.lower(
                    tostring(
                        Object.Text
                    )
                )
        end)

        if Score <= 0 then
            for Index, Name in ipairs(Names) do
                if ObjectText ~= ""
                    and string.find(
                        ObjectText,
                        Name,
                        1,
                        true
                    )
                then
                    Score =
                        math.max(
                            Score,
                            600 - Index * 15
                        )
                end
            end
        end

        -- Mobile Experience Controls often put the text in a child
        -- TextLabel rather than on the GuiButton itself. Read those labels
        -- so "Leaderboard" / "Inventory" can be found on touch devices.
        if Score <= 0 then
            for _, Child in ipairs(
                Object:GetDescendants()
            ) do

                if Child:IsA("TextLabel")
                    or Child:IsA("TextButton")
                    or Child:IsA("TextBox")
                then

                    local ChildText =
                        ""

                    pcall(function()
                        ChildText =
                            string.lower(
                                tostring(
                                    Child.Text
                                )
                            )
                    end)

                    if ChildText ~= "" then
                        for Index, Name in ipairs(Names) do
                            if string.find(
                                ChildText,
                                Name,
                                1,
                                true
                            )
                            then
                                Score =
                                    math.max(
                                        Score,
                                        800 - Index * 15
                                    )
                            end
                        end
                    end
                end
            end
        end

        if Score <= 0 then
            continue
        end

        local Ancestor =
            Object.Parent

        for _ = 1, 6 do

            if not Ancestor then
                break
            end

            local AncestorName =
                string.lower(
                    Ancestor.Name
                )

            if string.find(
                AncestorName,
                "topbar",
                1,
                true
            )
            or string.find(
                AncestorName,
                "top_bar",
                1,
                true
            )
            then

                Score +=
                    250

            end

            if string.find(
                AncestorName,
                "right",
                1,
                true
            )
            then

                Score +=
                    60

            end

            if string.find(
                AncestorName,
                "chrome",
                1,
                true
            )
            then

                Score +=
                    40

            end

            if string.find(
                AncestorName,
                "experience",
                1,
                true
            )
            or string.find(
                AncestorName,
                "hamburger",
                1,
                true
            )
            or string.find(
                AncestorName,
                "dropdown",
                1,
                true
            )
            or string.find(
                AncestorName,
                "menu",
                1,
                true
            )
            then

                Score +=
                    180

            end

            Ancestor =
                Ancestor.Parent
        end

        if Object.Visible then
            Score += 20
        end

        if Score > BestScore then
            BestScore =
                Score

            Best =
                Object
        end
    end

    NativeActionButtonCache[ActionName] =
        Best

    return Best
end

local function ActivateNativeActionButton(
    ActionName
)

    local Button =
        FindNativeActionButton(
            ActionName
        )

    if not Button then
        return false
    end

    PrepareButton(
        Button
    )

    local FireSignal =
        firesignal

    if type(FireSignal) ==
        "function"
    then

        local Fired =
            pcall(function()

                FireSignal(
                    Button.Activated
                )

            end)

        if Fired then
            return true
        end
    end

    local Activated =
        pcall(function()

            Button:Activate()

        end)

    return Activated
end

--============================================================
-- ROBUST MAIN BUTTON CLICK CATCHER
--============================================================
-- Some RBXM versions expose a named hit area that is a GuiButton but
-- can be underneath another transparent GUI object. A transparent
-- catcher on the actual action row gives the visible action a reliable
-- mouse/touch target without changing the artwork.
--============================================================

function CreateMainClickCatcher(
    Row,
    Name
)

    if not Row or not Row:IsA("GuiObject") then
        return nil
    end

    local Existing = Row:FindFirstChild(Name)

    if Existing and Existing:IsA("GuiButton") then
        PrepareButton(Existing)
        return Existing
    end

    local Button = Instance.new("ImageButton")

    Button.Name = Name
    Button.BackgroundTransparency = 1
    Button.BorderSizePixel = 0
    Button.Image = ""
    Button.ImageTransparency = 1
    Button.AutoButtonColor = false
    Button.Active = true
    Button.Selectable = true
    Button.ClipsDescendants = false
    Button.Size = UDim2.fromScale(1, 1)
    Button.Position = UDim2.fromScale(0, 0)
    Button.ZIndex = 10000
    Button.Parent = Row

    return Button
end

local function FindPlayerListMaster()

    local First = nil
    local VisibleBest = nil

    for _, Object in ipairs(
        CoreGui:GetDescendants()
    ) do

        local IsPlayerListName =
            Object.Name == "PlayerListMaster"
            or Object.Name == "PlayerList"
            or Object.Name == "PlayerListFrame"
            or Object.Name == "PlayerListContainer"

        if IsPlayerListName
            and Object:IsA("GuiObject")
        then

            local IsOurs =
                false

            pcall(function()
                IsOurs =
                    Object:IsDescendantOf(
                        TopBarApp
                    )
            end)

            if not IsOurs then

                if not First then
                    First = Object
                end

                local Visible = false
                pcall(function()
                    Visible = IsGuiTreeVisible(Object)
                end)

                if Visible then
                    VisibleBest = Object
                    break
                end
            end
        end
    end

    return VisibleBest or First
end

--============================================================
-- NATIVE STATE DETECTION
--============================================================

IsGuiTreeVisible = function(
    Object
)

    if not Object then
        return false
    end

    local Current = Object

    while Current do

        if Current:IsA("GuiObject") then

            local Success,
                Visible =
                pcall(function()
                    return Current.Visible
                end)

            if Success
                and not Visible
            then

                return false
            end
        end

        Current = Current.Parent
    end

    return true
end

local function FindBackpackInventoryFrame()

    local PreferredNames = {

        BackpackGui = true,
        Backpack = true,
        Inventory = true,
        InventoryFrame = true,
        BackpackFrame = true,
        BackpackContainer = true,
        InventoryContainer = true,
        BackpackMenu = true

    }

    for _, Object in ipairs(
        CoreGui:GetDescendants()
    ) do

        if Object:IsA("GuiObject") then

            local IsOurs = false

            pcall(function()
                IsOurs = Object:IsDescendantOf(TopBarApp)
            end)

            if not IsOurs then

                if PreferredNames[Object.Name]
                    and IsGuiTreeVisible(Object)
                then

                    return Object
                end
            end
        end
    end

    return nil
end

                    Name
                )

            end)

        if Success
            and type(Value) == "boolean"
        then

            return Value,
                Name
        end
    end

    return nil,
        nil
end

local PinRows = {}
local PinOverlay = nil

local function PositionPinButton(Button)

    if not Button or not Button.Parent then
        return
    end

    local Row = PinRows[Button]

    if not Row or not Row.Parent then
        Row = Button.Parent
    end

    if not Row then
        return
    end

    pcall(function()
        Button.AnchorPoint = Vector2.new(1, 0.5)
    end)

    -- Pin is intentionally independent of the row's layout.
    -- It sits at the far-right edge of the submenu and must never
    -- participate in the row UIListLayout or push the text.
    local Host = Button.Parent

    pcall(function()
        if Host and Host:IsA("GuiObject") then
            local HostAbs = Host.AbsolutePosition
            local RowAbs = Row.AbsolutePosition
            local RowHeight = Row.AbsoluteSize.Y
            local HostWidth = Host.AbsoluteSize.X
            local ButtonWidth = Button.AbsoluteSize.X

            local RelativeY =
                (RowAbs.Y - HostAbs.Y)
                + (RowHeight * 0.5)

            Button.Position =
                UDim2.fromOffset(
                    HostWidth - 18,
                    math.floor(RelativeY + 0.5)
                )
        end
    end)
end

local function WatchPinButtonPosition(Button)

    if not Button or not Button.Parent then
        return
    end

    local Bound = false
    pcall(function()
        Bound = Button:GetAttribute("__OldChrome2024PinPositionWatched") == true
    end)

    PositionPinButton(Button)

    if Bound then
        return
    end

    pcall(function()
        Button:SetAttribute("__OldChrome2024PinPositionWatched", true)
    end)

    local Row = PinRows[Button]

    if not Row then
        Row = Button.Parent
    end

    if Row then
        pcall(function()
            Row:GetPropertyChangedSignal("AbsolutePosition"):Connect(function()
                PositionPinButton(Button)
            end)
        end)

        pcall(function()
            Row:GetPropertyChangedSignal("AbsoluteSize"):Connect(function()
                PositionPinButton(Button)
            end)
        end)
    end

    pcall(function()
        if Button.Parent and Button.Parent:IsA("GuiObject") then
            Button.Parent:GetPropertyChangedSignal("AbsolutePosition"):Connect(function()
                PositionPinButton(Button)
            end)

            Button.Parent:GetPropertyChangedSignal("AbsoluteSize"):Connect(function()
                PositionPinButton(Button)
            end)
        end
    end)

    pcall(function()
        if ScrollingFrame then
            ScrollingFrame:GetPropertyChangedSignal("CanvasPosition"):Connect(function()
                PositionPinButton(Button)
            end)
        end
    end)
end

RestoreAllPinGlyphs = function()

    for ActionName, Button in pairs(_G.__OldChrome2024PinButtons or {}) do
        if Button and Button.Parent then

            pcall(function()
                Button.Visible = true
                Button.Active = true
                Button.ClipsDescendants = false
                Button.ImageTransparency = 1
                Button.ZIndex = 1000
            end)

            PositionPinButton(Button)

            local Glyph = Button:FindFirstChild("PinGlyph")
            if Glyph then
                pcall(function()
                    Glyph.Visible = true
                    Glyph.ImageTransparency = 0
                    Glyph.ZIndex = 2001
                end)

                ApplyPinGlyphState(
                    Glyph,
                    _G.__OldChrome2024PinnedActions[ActionName] == true
                )
            end
        end
    end
end

_G.__OldChrome2024RestorePinGlyphs = RestoreAllPinGlyphs

local function ReadNativeChatState()

    if IsActionLocked("chat") then
        return ChatOpen
    end

    -- ChatActive is the authoritative CoreScripts state when exposed.
    local CoreState =
        ReadGetCoreBoolean({
            "ChatActive"
        })

    if CoreState ~= nil then
        return CoreState
    end

    local Success,
        Focused =
        pcall(function()

            return
                TextChatService
                    .ChatInputBarConfiguration
                    .IsFocused

        end)

    if Success
        and Focused
    then

        return true
    end

    local WindowSuccess,
        WindowEnabled =
        pcall(function()

            return
                TextChatService
                    .ChatWindowConfiguration
                    .Enabled

        end)

    if WindowSuccess
        and type(WindowEnabled) == "boolean"
    then

        return WindowEnabled
    end

    return ChatOpen
end

local function ReadNativeInventoryState(IgnoreLock)

    if not IgnoreLock and IsActionLocked("backpack") then
        return InventoryOpen
    end

    if os.clock() < (NativeHotkeyOverrideUntil.backpack or 0) then
        return InventoryOpen
    end

    -- Prefer the actual native GUI when it exists. This catches the
    -- physical Backquote toggle even when GetCore still has its old value.
    local InventoryFrame =
        FindBackpackInventoryFrame()

    if InventoryFrame then

        local Success,
            Visible =
            pcall(function()
                return IsGuiTreeVisible(InventoryFrame)
            end)

        if Success
            and type(Visible) == "boolean"
        then
            return Visible
        end
    end

    -- GetCore is the fallback for clients where the native backpack
    -- does not expose a stable visible GuiObject.
    local CoreState =
        ReadGetCoreBoolean({
            "BackpackActive",
            "BackpackOpen",
            "BackpackIsOpen"
        })

    if CoreState ~= nil then
        return CoreState
    end

    return InventoryOpen
end

local function ReadNativePlayerListState(IgnoreLock)

    if not IgnoreLock and IsActionLocked("leaderboard") then
        return LeaderboardOpen
    end

    if os.clock() < (NativeHotkeyOverrideUntil.leaderboard or 0) then
        return LeaderboardOpen
    end

    -- GetCore is the authoritative CoreScript state for PlayerList on the
    -- clients this replacement is targeting. Use the GUI tree only as a
    -- fallback because PlayerListMaster can remain mounted while hidden.
    local CoreState =
        ReadGetCoreBoolean({
            "PlayerListActive",
            "PlayerListOpen",
            "PlayerListIsOpen"
        })

    if CoreState ~= nil then
        return CoreState
    end

    local PlayerListMaster = FindPlayerListMaster()

    if PlayerListMaster then

        local Success, Visible = pcall(function()
            return IsGuiTreeVisible(PlayerListMaster)
        end)

        if Success and type(Visible) == "boolean" then
            return Visible
        end

    end

    return LeaderboardOpen
end

local function ReadNativeEmotesState(IgnoreLock)

    if not IgnoreLock and IsActionLocked("emotes") then
        return EmotesOpen
    end

    -- GuiService is the actual menu state. GetCore can report the previous
    -- state for a short time after a click, which caused the two-click bug.
    local Success,
        Open =
        pcall(function()

            return GuiService:GetEmotesMenuOpen()

        end)

    if Success
        and type(Open) == "boolean"
    then

        return Open
    end

    local CoreState =
        ReadGetCoreBoolean({
            "EmotesMenuActive",
            "EmotesActive",
            "EmotesMenuOpen",
            "EmotesOpen"
        })

    if CoreState ~= nil then
        return CoreState
    end

    return EmotesOpen
end

    CoreType
)

        return true
    end

        CoreType
    )

local function SyncCoreGuiAvailability()

    local AllDisabled =
        GetCoreGuiAllDisabled()

    local ChatVisible =
        ShouldShowForCoreType(
            Enum.CoreGuiType.Chat
        )

    local PlayerListVisible =
        ShouldShowForCoreType(
            Enum.CoreGuiType.PlayerList
        )

    local BackpackVisible =
        ShouldShowForCoreType(
            Enum.CoreGuiType.Backpack
        )

    local EmotesVisible =
        ShouldShowForCoreType(
            Enum.CoreGuiType.EmotesMenu
        )

    SetVisible(MenuIconHolder, true)
    SetVisible(ChromeToggle, true)
    SetVisible(NineDot, ChromeOpen)
    SetVisible(Trust, ChromeOpen)
    SetVisible(Chat, ChromeOpen)
    SetVisible(Leaderboard, ChromeOpen)

    -- Our replacement Backpack/Emotes icons are always present while the
    -- Chrome is expanded. CoreGui Enabled must never delete them.
    if InventoryRoot then
        SetVisible(InventoryRoot, ChromeOpen)
    end

    if EmotesRoot then
        SetVisible(EmotesRoot, ChromeOpen)
    end

    -- Keep every submenu action present; GetCore is the state reader.
    SetVisible(SubChat, true)
    SetVisible(SubLeaderboard, true)
    SetVisible(SubBackpack, true)
    SetVisible(SubEmotes, true)
    SetVisible(SubReport, true)
    SetVisible(SubNineDot, true)

    local Key =
        tostring(AllDisabled) .. "|"
        .. tostring(ChatVisible) .. "|"
        .. tostring(PlayerListVisible) .. "|"
        .. tostring(BackpackVisible) .. "|"
        .. tostring(EmotesVisible)

    if CoreAvailabilityLast ~= Key then
        CoreAvailabilityLast = Key

        if VERBOSE then
            print(
                string.format(
                    "[OLD CHROME][VERBOSE] CoreGui = AllDisabled=%s | Chat=%s | PlayerList=%s | Backpack=%s | Emotes=%s",
                    tostring(AllDisabled),
                    tostring(ChatVisible),
                    tostring(PlayerListVisible),
                    tostring(BackpackVisible),
                    tostring(EmotesVisible)
                )
            )
        end
    end
end

-- STATE SYNC
--============================================================

local function SyncNativeState(ForceNativeRead)

    local NewChatOpen =
        ReadNativeChatState()

    local NewLeaderboardOpen

    -- Mobile opens PlayerList through the Experience Controls menu, not
    -- the desktop Tab shortcut. On touch devices the native GUI visibility
    -- therefore already represents the real open state and must not use
    -- the desktop inversion used by the previous V21 fix.
    if UserInputService.TouchEnabled
        and not UserInputService.KeyboardEnabled
    then

        NewLeaderboardOpen =
            ReadNativePlayerListState(
                ForceNativeRead == true
            )

    else

        NewLeaderboardOpen =
            not ReadNativePlayerListState(
                ForceNativeRead == true
            )

    end

    -- Backpack open/closed is tracked from the real Backquote input
    -- and from our synthetic Backquote click path. Do not overwrite it
    -- with a stale/ambiguous native GUI scan here.
    local NewInventoryOpen =
        InventoryOpen

    local NewEmotesOpen =
        ReadNativeEmotesState()

    if not IsActionLocked("chat") then
        ChatOpen = NewChatOpen
    end

    if not IsActionLocked("leaderboard") then
        LeaderboardOpen = NewLeaderboardOpen
    end

    if not IsActionLocked("backpack") then
        InventoryOpen = NewInventoryOpen
    end

    if not IsActionLocked("emotes") then
        EmotesOpen = NewEmotesOpen
    end

    UpdateMainIconStates()
    SyncCoreGuiAvailability()
    RestoreAllPinGlyphs()

    if type(
        _G.__OldChrome2024SyncPinned
    ) == "function"
    then

        _G.__OldChrome2024SyncPinned()
    end
end

--============================================================
-- NATIVE HOTKEY STATE WATCH
--============================================================

local function SyncAfterNativeHotkey()
    -- Respect the short local-state lock; do not force-read stale native state.
    task.defer(function()
        SyncNativeState()
    end)
end

UserInputService.InputBegan:Connect(
    function(
        Input,
        GameProcessed
    )

        if Input.KeyCode ==
            Enum.KeyCode.Period
        then

            EmotesOpen =
                not EmotesOpen

            SetActionLocalState(
                "emotes",
                EmotesOpen,
                0.75
            )

            NativeHotkeyOverrideUntil.emotes =
                os.clock() + 0.75

            UpdateMainIconStates()
            SyncAfterNativeHotkey()

        elseif Input.KeyCode ==
            Enum.KeyCode.Escape
        then

            task.defer(SyncNativeState)

        elseif Input.KeyCode ==
            Enum.KeyCode.Backquote
        then

            if os.clock() < SuppressSyntheticBackquoteUntil then

                -- This is the Backquote generated by our custom
                -- Backpack button. The click already applied the desired
                -- state, so do not toggle it a second time.
                SuppressSyntheticBackquoteUntil = 0

                return
            end

            InventoryOpen = not InventoryOpen
            NativeHotkeyOverrideUntil.backpack = os.clock() + 0.75
            SetActionLocalState("backpack", InventoryOpen, 0.75)
            UpdateMainIconStates()
            SyncAfterNativeHotkey()

        elseif Input.KeyCode ==
            Enum.KeyCode.Tab
        then

            if os.clock() < SuppressSyntheticTabUntil then

                SuppressSyntheticTabUntil = 0

                return
            end

            LeaderboardOpen = not LeaderboardOpen
            NativeHotkeyOverrideUntil.leaderboard = os.clock() + 0.75
            SetActionLocalState("leaderboard", LeaderboardOpen, 0.75)
            UpdateMainIconStates()
            SyncAfterNativeHotkey()

        elseif Input.KeyCode ==
            Enum.KeyCode.Slash
        then

            ReopenChatFromSlash()

            ChatOpen = true

            SetActionLocalState(
                "chat",
                true,
                0.8
            )

            NativeHotkeyOverrideUntil.chat =
                os.clock() + 0.8

            UpdateMainIconStates()

            task.defer(function()
                ReopenChatFromSlash()
            end)
        end

    end
)

UserInputService.InputEnded:Connect(
    function(Input)
        -- Tab/backquote state is predicted on InputBegan; do not issue
        -- another read here because the native key release can race it.
    end
)

local function SetupOptionalIcon(
    Root,
    HitArea,
    NormalHighlight,
    SelectedHighlight,
    IsOpen,
    PersistentSelected
)

    if not HitArea
        or not HitArea:IsA("GuiButton")
    then

        return

    end

    PrepareButton(
        HitArea
    )

    HideHighlight(
        NormalHighlight
    )

    HideHighlight(
        SelectedHighlight
    )

    HitArea.MouseEnter:Connect(
        function()

            ShowHighlight(
                NormalHighlight
            )

            if not (
                PersistentSelected
                and IsOpen()
            ) then

                HideHighlight(
                    SelectedHighlight
                )

            end
        end
    )

    HitArea.MouseLeave:Connect(
        function()

            HideHighlight(
                NormalHighlight
            )

            if PersistentSelected
                and IsOpen()
            then

                ShowHighlight(
                    SelectedHighlight,
                    0.9
                )

            else

                HideHighlight(
                    SelectedHighlight
                )

            end
        end
    )
end

--============================================================
-- LEADERBOARD TOGGLE
--============================================================

local function ToggleLeaderboardNative()

    local Desired =
        not LeaderboardOpen

    LeaderboardOpen =
        Desired

    NativeHotkeyOverrideUntil.leaderboard =
        os.clock() + 0.75

    SetActionLocalState(
        "leaderboard",
        Desired,
        0.75
    )

    UpdateMainIconStates()

    -- Mobile has no Tab shortcut. Use the real Experience Controls
    -- PlayerList action when it is mounted, then fall back to the native
    -- PlayerList GUI itself. This opens the same Roblox PlayerList rather
    -- than making a separate/custom leaderboard.
    if UserInputService.TouchEnabled
        and not UserInputService.KeyboardEnabled
    then

        -- Some current clients expose the native Experience Controls
        -- action through StarterGui:SetCore even though it is not a public
        -- API. Try it first when the client provides it.
        local CoreActivated =
            SetGetCoreBoolean(
                {
                    "PlayerListActive",
                    "PlayerListOpen",
                    "PlayerListIsOpen"
                },
                Desired
            )

        if CoreActivated then
            return
        end

        if ActivateNativeActionButton(
            "leaderboard"
        )
        then
            return
        end

        local PlayerListFrame =
            FindPlayerListMaster()

        if PlayerListFrame then
            pcall(function()
                PlayerListFrame.Visible =
                    Desired
            end)

            return
        end

        pcall(function()
            StarterGui:SetCoreGuiEnabled(
                Enum.CoreGuiType.PlayerList,
                Desired
            )
        end)

        return
    end

    -- Desktop: exactly one native Tab action. The synthetic Tab must not
    -- re-toggle our local state in the InputBegan handler above.
    SuppressSyntheticTabUntil =
        os.clock() + 0.35

    SendTab()
end

ToggleLeaderboard =
    ToggleLeaderboardNative

--============================================================
-- BACKPACK NATIVE TOGGLE
--============================================================

local function SetNativeBackpack(
    Desired
)

    -- Mobile has no Backquote shortcut. First activate the real Inventory
    -- item from Roblox's native Experience Controls.
    if UserInputService.TouchEnabled
        and not UserInputService.KeyboardEnabled
    then

        local CoreActivated =
            SetGetCoreBoolean(
                {
                    "BackpackActive",
                    "BackpackOpen",
                    "BackpackIsOpen"
                },
                Desired
            )

        if CoreActivated then
            return true
        end

        if ActivateNativeActionButton(
            "backpack"
        )
        then
            return true
        end

        -- Last native fallback: some clients expose the actual Backpack
        -- inventory frame directly in CoreGui. Visibility changes here are
        -- still acting on Roblox's own Backpack UI, not our replacement.
        local BackpackFrame =
            FindBackpackInventoryFrame()

        if BackpackFrame then
            pcall(function()
                BackpackFrame.Visible =
                    Desired
            end)

            return true
        end

        return pcall(function()
            StarterGui:SetCoreGuiEnabled(
                Enum.CoreGuiType.Backpack,
                Desired
            )
        end)
    end

    -- Desktop fallback remains the native Backquote toggle.
    if ActivateNativeActionButton(
        "backpack"
    )
    then

        return true
    end

    SuppressSyntheticBackquoteUntil =
        os.clock() + 0.35

    SendBackquote()

    return true
end

local function FindReportButton()

    if not RobloxGui then
        return nil
    end

    local Candidates =
        {}

    for _, Object in ipairs(
        RobloxGui:GetDescendants()
    ) do

        if not Object:IsA("GuiButton") then
            continue
        end

        if not Object.Visible then
            continue
        end

        local Score =
            0

        local Name =
            string.lower(
                Object.Name
            )

        if Name ==
            "report"
        then

            Score +=
                200

        elseif string.find(
            Name,
            "report",
            1,
            true
        )
        then

            Score +=
                100

        end

        if Object:IsA(
            "TextButton"
        )
        then

            local Text =
                string.lower(
                    Object.Text
                )

            if Text ==
                "report"
            then

                Score +=
                    200

            elseif string.find(
                Text,
                "report",
                1,
                true
            )
            then

                Score +=
                    100

            end
        end

        for _, Child in ipairs(
            Object:GetDescendants()
        ) do

            if Child:IsA(
                "TextLabel"
            )
            or Child:IsA(
                "TextButton"
            )
            then

                local Text =
                    string.lower(
                        Child.Text
                    )

                if Text ==
                    "report"
                then

                    Score +=
                        150

                elseif string.find(
                    Text,
                    "report",
                    1,
                    true
                )
                then

                    Score +=
                        75

                end
            end
        end

        if Score > 0 then

            table.insert(
                Candidates,
                {
                    Object =
                        Object,

                    Score =
                        Score
                }
            )

        end
    end

    table.sort(
        Candidates,
        function(A, B)

            return A.Score >
                B.Score

        end
    )

    return Candidates[1]
        and Candidates[1].Object
end

local function ActivateGuiButton(
    Button
)

    if not Button then
        return false
    end

    local Success =
        pcall(function()

            Button:Activate()

        end)

    if Success then
        return true
    end

    if type(firesignal) ==
        "function"
    then

        local Signal

        pcall(function()

            if Button:IsA(
                "TextButton"
            )
            or Button:IsA(
                "ImageButton"
            )
            then

                Signal =
                    Button.MouseButton1Click

            end

        end)

        if Signal then

            return pcall(
                firesignal,
                Signal
            )

        end
    end

    return false
end

local function TryOpenReport()

    if TrustReportRunning then
        return
    end

    TrustReportRunning =
        true

    task.spawn(
        function()

            local StartTime =
                os.clock()

            local ReportButton

            repeat

                ReportButton =
                    FindReportButton()

                if not ReportButton then

                    task.wait(
                        0.05
                    )

                end

            until ReportButton
                or os.clock() -
                    StartTime >= 5

            if ReportButton then

                print(
                    "[OLD CHROME] Report button:",
                    ReportButton:GetFullName()
                )

                ActivateGuiButton(
                    ReportButton
                )

            else

                warn(
                    "[OLD CHROME] Could not locate the real Roblox Report button."
                )

            end

            TrustReportRunning =
                false

        end
    )
end

FixedUnibarOrder = {

    chrome_toggle =
        1,

    nine_dot =
        2,

    inventory =
        3,

    emotes =
        4,

    chat =
        5,

    leaderboard =
        6,

    trust_and_safety =
        7,

    div1 =
        8
}

function ShiftUDim2(
    Position,
    XOffset
)

    return UDim2.new(

        Position.X.Scale,
        Position.X.Offset +
            XOffset,

        Position.Y.Scale,
        Position.Y.Offset

    )
end

function GetOrderedOptionalIcons()

    local Result =
        {}

    -- HARD ORDER:
    -- Backpack FIRST.
    if InventoryRoot
        and InventoryRoot.Parent ==
            Frame3
    then

        table.insert(
            Result,
            InventoryRoot
        )

    end

    -- Emotes SECOND.
    if EmotesRoot
        and EmotesRoot.Parent ==
            Frame3
    then

        table.insert(
            Result,
            EmotesRoot
        )

    end

    return Result
end

UnibarLayoutDetached = false

function DetachDirectUnibarLayouts()

    if UnibarLayoutDetached or not Frame3 then
        return
    end

    for _, Layout in ipairs(ExistingLayouts) do
        if Layout and Layout.Parent == Frame3 then
            pcall(function()
                Layout.Parent = nil
            end)
        end
    end

    UnibarLayoutDetached = true
end

RepositionUnibar = function()

    if not Frame3 then
        return
    end

    DetachDirectUnibarLayouts()

    local OptionalCount =
        0

    if InventoryRoot
        and InventoryRoot.Parent ==
            Frame3
    then

        OptionalCount +=
            1

    end

    if EmotesRoot
        and EmotesRoot.Parent ==
            Frame3
    then

        OptionalCount +=
            1

    end

    --========================================================
    -- LayoutOrder.
    --========================================================

    for _, Object in ipairs(
        Frame3:GetChildren()
    ) do

        if not Object:IsA("GuiObject") then
            continue
        end

        local Name =
            string.lower(
                Object.Name
            )

        local Order =
            FixedUnibarOrder[
                Name
            ]

        if Order then

            pcall(function()

                Object.LayoutOrder =
                    Order

            end)

        end
    end

    --========================================================
    -- Physical-position fallback.
    --========================================================

    for Name, OriginalPosition in pairs(
        OriginalUnibarPositions
    ) do

        local Object =
            Frame3:FindFirstChild(
                Name
            )

        if not Object then
            continue
        end

        local LowerName =
            string.lower(Name)

        if LowerName ==
            "chat"
            or LowerName ==
                "leaderboard"
            or LowerName ==
                "trust_and_safety"
            or LowerName ==
                "div1"
        then

            pcall(function()

                Object.Position =
                    ShiftUDim2(
                        OriginalPosition,
                        OptionalCount * 44
                    )

            end)

        end
    end

    --========================================================
    -- Optional positions.
    --
    -- Backpack ALWAYS first.
    -- Emotes ALWAYS second.
    --========================================================

    local ChatPosition =
        OriginalUnibarPositions.chat

    if ChatPosition then

        local OptionalIndex =
            0

        for _, Object in ipairs(
            GetOrderedOptionalIcons()
        ) do

            OptionalIndex +=
                1

            pcall(function()

                Object.Position =
                    ShiftUDim2(
                        ChatPosition,
                        (
                            OptionalIndex - 1
                        ) * 44
                    )

            end)
        end
    end

    --========================================================
    -- EXPLICIT TRUST MOVE.
    --========================================================

    if Trust
        and OriginalUnibarPositions[
            "trust_and_safety"
        ]
    then

        pcall(function()

            Trust.Position =
                ShiftUDim2(
                    OriginalUnibarPositions[
                        "trust_and_safety"
                    ],
                    OptionalCount *
                        44
                )

        end)

    end

    --========================================================
    -- EXPLICIT DIV1 MOVE.
    --
    -- THIS IS THE IMPORTANT PART:
    -- DIV1 is physically shifted, not only LayoutOrder.
    --========================================================

    if DIV1
        and OriginalUnibarPositions[
            "DIV1"
        ]
    then

        pcall(function()

            DIV1.Position =
                ShiftUDim2(
                    OriginalUnibarPositions[
                        "DIV1"
                    ],
                    OptionalCount *
                        44
                )

        end)

    end

    -- Some RBXMs can lowercase names differently.
    -- Explicit direct object fallback.
    if DIV1 then

        local Base =
            OriginalUnibarPositions[
                DIV1.Name
            ]

        if Base then

            pcall(function()

                DIV1.Position =
                    ShiftUDim2(
                        Base,
                        OptionalCount *
                            44
                    )

            end)

        end
    end
end

UpdateChromeWidth = function()

    if not Frame2 then
        return
    end

    local Count =
        0

    if InventoryRoot
        and InventoryRoot.Parent ==
            Frame3
    then

        Count +=
            1

    end

    if EmotesRoot
        and EmotesRoot.Parent ==
            Frame3
    then

        Count +=
            1

    end

    local Width

    if ChromeOpen then

        Width =
            225 +
            Count * 44

    else

        local PinnedCount =
            0

        if _G.__OldChrome2024PinnedOrder
            and _G.__OldChrome2024PinnedActions
        then

            for _, ActionName in ipairs(
                _G.__OldChrome2024PinnedOrder
            ) do

                if _G.__OldChrome2024PinnedActions[
                    ActionName
                ]
                then

                    PinnedCount +=
                        1
                end
            end
        end

        Width =
            44 +
            PinnedCount * 44
    end

    Tween(
        Frame2,
        ChromeTweenInfo,
        {
            Size =
                UDim2.fromOffset(
                    Width,
                    44
                )
        }
    )
end

function OldChrome2024CreatePinnedIcon(
    ActionName
)

    local Existing =
        _G.__OldChrome2024PinnedRoots[
            ActionName
        ]

    if Existing
        and Existing.Parent
    then

        local ExistingIcon = Existing:FindFirstChild("PinnedIcon")
        if ExistingIcon then
            pcall(function()
                Existing.Visible = not ChromeOpen
                ExistingIcon.Visible = true
                ExistingIcon.ImageTransparency = 0
            end)
        end
        return Existing
    end

    local Source =
        OldChrome2024GetPinnedSource(
            ActionName
        )

    if not Source
        or not Frame2
    then

        return nil
    end

    local SourceIcon =
        Path(
            Source,
            "IntegrationIconFrame",
            "IntegrationIcon"
        )

    if not SourceIcon then
        return nil
    end

    --========================================================
    -- CRITICAL PIN RENDER FIX
    --
    -- Clone the already-loaded source artwork instead of creating
    -- a fresh ImageButton whose image can take a separate render
    -- pass to appear. A separate transparent hitbox handles input.
    --========================================================

    local Root =
        Instance.new(
            "Frame"
        )

    Root.Name =
        "pinned_" ..
        ActionName

    Root.BackgroundTransparency =
        1

    Root.BorderSizePixel =
        0

    Root.Size =
        UDim2.fromOffset(
            44,
            44
        )

    Root.ZIndex =
        10000

    pcall(function()
        Root:SetAttribute(
            "__OldChrome2024Pinned",
            true
        )
    end)

    Root.Parent =
        Frame2

    local Icon =
        SourceIcon:Clone()

    Icon.Name =
        "PinnedIcon"

    Icon.AnchorPoint =
        Vector2.new(
            0.5,
            0.5
        )

    Icon.Position =
        UDim2.fromScale(
            0.5,
            0.5
        )

    Icon.Size =
        UDim2.fromOffset(
            36,
            36
        )

    Icon.BackgroundTransparency =
        1

    pcall(function()
        Icon.BorderSizePixel = 0
    end)

    Icon.ZIndex =
        10001

    Icon.Visible =
        true

    pcall(function()
        Icon.ImageTransparency = 0
    end)

    pcall(function()
        Icon.ImageRectSize =
            Vector2.new(
                36,
                36
            )
    end)

    Icon.Parent =
        Root

    if ActionName ==
        "backpack"
    then

        ApplyBackpackIconState(
            Icon,
            InventoryOpen
        )

    elseif ActionName ==
        "emotes"
    then

        ApplyEmotesIconState(
            Icon,
            EmotesOpen
        )

    elseif ActionName ==
        "leaderboard"
    then

        SetRect(
            Icon,
            0,
            LeaderboardOpen
            and 380
            or 114
        )

    end

    -- Independent input layer. This prevents the visual clone from
    -- being replaced by a transparent ImageButton and guarantees the
    -- pinned icon reacts immediately.
    local Hitbox =
        Instance.new(
            "ImageButton"
        )

    Hitbox.Name =
        "PinnedIconHitbox"

    Hitbox.BackgroundTransparency =
        1

    Hitbox.BorderSizePixel =
        0

    Hitbox.Image =
        ""

    Hitbox.ImageTransparency =
        1

    Hitbox.AutoButtonColor =
        false

    Hitbox.Active =
        true

    Hitbox.Selectable =
        true

    Hitbox.Size =
        UDim2.fromScale(
            1,
            1
        )

    Hitbox.Position =
        UDim2.fromScale(
            0,
            0
        )

    Hitbox.ZIndex =
        10002

    Hitbox.Parent =
        Root

    _G.__OldChrome2024PinnedRoots[
        ActionName
    ] =
        Root

    local PinnedBusyUntil = 0

    local function ActivatePinned()

        local Now = os.clock()

        if Now < PinnedBusyUntil then
            return
        end

        PinnedBusyUntil = Now + 0.12

        if ActionName ==
            "backpack"
        then

            ToggleInventory()

        elseif ActionName ==
            "emotes"
        then

            ToggleEmotes()

        elseif ActionName ==
            "chat"
        then

            ToggleChat()

        elseif ActionName ==
            "leaderboard"
        then

            ToggleLeaderboard()

        elseif ActionName ==
            "report"
        then

            ToggleTrust()

        elseif ActionName ==
            "nine_dot"
        then

            if SubMenuOpen then
                CloseSubmenu()
            else
                OpenSubmenu()
            end
        end
    end

    pcall(function()
        Hitbox.Activated:Connect(
            ActivatePinned
        )
    end)

    pcall(function()
        Hitbox.MouseButton1Click:Connect(
            ActivatePinned
        )
    end)

    Root.Visible =
        not ChromeOpen

    pcall(function()
        Icon.Visible = true
        Icon.ImageTransparency = 0
        Hitbox.Visible = true
    end)

    return Root
end

    ActionName
)

    local Root =
        _G.__OldChrome2024PinnedRoots[
            ActionName
        ]

    if Root then

        pcall(function()
            Root:Destroy()
        end)
    end

    _G.__OldChrome2024PinnedRoots[
        ActionName
    ] =
        nil
end

function OldChrome2024PinnedAvailable(
    ActionName
)
    return true
end

function OldChrome2024RepositionPinned()

    if not Frame2 then
        return
    end

    -- Visibility is applied before position/size work so a pinned icon
    -- cannot wait for a later layout pass to become visible.
    if not ChromeOpen then
        for _, ActionName in ipairs(
            _G.__OldChrome2024PinnedOrder
        ) do
            if _G.__OldChrome2024PinnedActions[ActionName] then
                local Root =
                    _G.__OldChrome2024PinnedRoots[ActionName]
                if Root and Root.Parent == Frame2 then
                    Root.Visible = true
                end
            end
        end
    end

    local Index =
        0

    for _, ActionName in ipairs(
        _G.__OldChrome2024PinnedOrder
    ) do

        if _G.__OldChrome2024PinnedActions[
            ActionName
        ]
        then

            local Root =
                _G.__OldChrome2024PinnedRoots[
                    ActionName
                ]

            if Root
                and Root.Parent ==
                    Frame2
            then

                Index +=
                    1

                Root.Position =
                    UDim2.fromOffset(
                        44 +
                            (Index - 1) *
                            44,
                        0
                    )

                local Visible =
                    not ChromeOpen
                    and
                    OldChrome2024PinnedAvailable(
                        ActionName
                    )

                Root.Visible =
                    Visible
            end
        end
    end
end

function OldChrome2024SyncPinned()

    -- Every current pinnable action has a canonical open-unibar source.
    -- Pinning must never make that source disappear while Chrome is open.
    if ChromeOpen and Frame3 then
        for _, ActionName in ipairs(_G.__OldChrome2024PinnedOrder) do
            if _G.__OldChrome2024PinnedActions[ActionName] then
                local Source = OldChrome2024GetPinnedSource(ActionName)
                if Source and Source.Parent == Frame3 then
                    pcall(function()
                        Source.Visible = true
                    end)
                end
            end
        end
    end

    for _, ActionName in ipairs(
        _G.__OldChrome2024PinnedOrder
    ) do

        if _G.__OldChrome2024PinnedActions[
            ActionName
        ]
        then

            OldChrome2024CreatePinnedIcon(
                ActionName
            )
        end
    end

    for ActionName, Root in pairs(
        _G.__OldChrome2024PinnedRoots
    ) do

        if not Root
            or not Root.Parent
        then
            continue
        end

        local Icon =
            Root:FindFirstChild(
                "PinnedIcon"
            )

        if not Icon then
            continue
        end

        if ActionName ==
            "chat"
        then

            pcall(function()
                Icon.Image =
                    ChatIcon.Image
            end)

            SetRect(
                Icon,
                ChatOpen
                and 152
                or 418,
                ChatOpen
                and 114
                or 38
            )

        elseif ActionName ==
            "leaderboard"
        then

            pcall(function()
                Icon.Image =
                    LeaderboardIcon.Image
            end)

            SetRect(
                Icon,
                0,
                LeaderboardOpen
                and 380
                or 114
            )

        elseif ActionName ==
            "backpack"
        then

            ApplyBackpackIconState(
                Icon,
                InventoryOpen
            )

        elseif ActionName ==
            "emotes"
        then

            ApplyEmotesIconState(
                Icon,
                EmotesOpen
            )

        elseif ActionName ==
            "report"
        then

            pcall(function()
                Icon.Image =
                    TrustIcon.Image
            end)

            SetRect(
                Icon,
                TrustActive
                and 266
                or 304,
                TrustActive
                and 76
                or 38
            )

        elseif ActionName ==
            "nine_dot"
        then

            local Source =
                Path(
                    NineDot,
                    "IntegrationIconFrame",
                    "IntegrationIcon"
                )

            if Source then

                pcall(function()

                    Icon.Image =
                        Source.Image

                    Icon.ImageRectOffset =
                        Source.ImageRectOffset

                end)
            end
        end

        pcall(function()

            Icon.Visible = true

            if ActionName ==
                "emotes"
            then

                Icon.ImageRectSize =
                    EMOTES_RECT_SIZE

            else

                Icon.ImageRectSize =
                    Vector2.new(
                        36,
                        36
                    )

            end

            Icon.ImageTransparency =
                0

        end)
    end

    OldChrome2024RepositionPinned()
end

_G.__OldChrome2024SyncPinned =
    OldChrome2024SyncPinned

function OldChrome2024CreatePinButton(
    Row,
    ActionName
)

    if not Row then
        return nil
    end

    local Existing =
        Row:FindFirstChild(
            "PinButton"
        )

    if Existing
        and Existing:IsA(
            "ImageButton"
        )
    then

        pcall(function()
            Existing.Visible = true
            Existing.Active = true
            Existing.ClipsDescendants = false
            Existing.Image = ""
            Existing.ImageTransparency = 1
            Existing.ZIndex = 1000
        end)

        SetPinGlyph(
            Existing,
            _G.__OldChrome2024PinnedActions[
                ActionName
            ] == true
        )

        PinRows[Existing] = Row

        if PinOverlay and Existing.Parent ~= PinOverlay then
            Existing.Parent = PinOverlay
        end

        WatchPinButtonPosition(Existing)

        _G.__OldChrome2024PinButtons[
            ActionName
        ] =
            Existing

        return Existing
    end

    local Button =
        Instance.new(
            "ImageButton"
        )

    Button.Name =
        "PinButton"

    -- 36x36 hit area.
    Button.Size =
        UDim2.fromOffset(
            36,
            36
        )

    Button.AnchorPoint =
        Vector2.new(
            1,
            0.5
        )

    Button.Position =
        UDim2.new(
            1,
            -4,
            0.5,
            0
        )

    Button.BackgroundTransparency =
        1

    Button.BorderSizePixel =
        0

    -- The Button itself must NOT render an icon.
    Button.Image =
        ""

    Button.ImageTransparency =
        1

    Button.AutoButtonColor =
        false

    Button.Active =
        true

    Button.Visible =
        true

    Button.ClipsDescendants =
        false

    Button.ZIndex =
        1000

    PinRows[Button] = Row

    Button.Parent =
        PinOverlay or Row

    WatchPinButtonPosition(Button)

    _G.__OldChrome2024PinButtons[
        ActionName
    ] =
        Button

    SetPinGlyph(
        Button,
        false
    )

    Button.Activated:Connect(
        function()

            local NewState =
                not (
                    _G.__OldChrome2024PinnedActions[
                        ActionName
                    ] == true
                )

            _G.__OldChrome2024PinnedActions[
                ActionName
            ] =
                NewState

            if NewState then

                local Exists =
                    false

                for _, ExistingName in ipairs(
                    _G.__OldChrome2024PinnedOrder
                ) do

                    if ExistingName ==
                        ActionName
                    then

                        Exists =
                            true

                        break
                    end
                end

                if not Exists then

                    table.insert(
                        _G.__OldChrome2024PinnedOrder,
                        ActionName
                    )
                end

                OldChrome2024CreatePinnedIcon(
                    ActionName
                )

            else

                for Index, ExistingName in ipairs(
                    _G.__OldChrome2024PinnedOrder
                ) do

                    if ExistingName ==
                        ActionName
                    then

                        table.remove(
                            _G.__OldChrome2024PinnedOrder,
                            Index
                        )

                        break
                    end
                end

                OldChrome2024DestroyPinnedIcon(
                    ActionName
                )
            end

            SetPinGlyph(
                Button,
                NewState
            )

            OldChrome2024SyncPinned()
            OldChrome2024RepositionPinned()
            RepositionUnibar()
            UpdateChromeWidth()

        end
    )

    return Button
end
