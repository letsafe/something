	if not ensureBubbleScreenGui() then return false end

		if changedType == Enum.CoreGuiType.Chat then bubbleChatCoreGuiAllowed = changedEnabled == true end
		if changedType == Enum.CoreGuiType.All then bubbleChatAllAllowed = changedEnabled == true end
		bubbleGuiAllowed = bubbleChatCoreGuiAllowed and bubbleChatAllAllowed
		local allowed = bubbleGuiAllowed

			if coreGuiType == Enum.CoreGuiType.Chat or coreGuiType == Enum.CoreGuiType.All then

			if not events or connectedFolders[events] then return end
			connectedFolders[events] = true
			local connectedRemotes = {}
			local function connectRemote(remote)
				if not remote or not remote:IsA("RemoteEvent") or connectedRemotes[remote] then return end
				if remote:GetAttribute("YekuClientPlaceholder") == true then return end
				if remote.Name == "OnNewMessage" then
					connectedRemotes[remote] = true
					addMessageConnection(remote.OnClientEvent, onUnfilteredLegacyMessage)

					connectedRemotes[remote] = true
					addMessageConnection(remote.OnClientEvent, onFilteredLegacyMessage)
				end
			end

			connectRemote(events:FindFirstChild("OnMessageDoneFiltering"))

			if child.Name == "DefaultChatSystemChatEvents" then connectLegacyFolder(child) end

-- ============================================================
-- PATH RESOLUTION
-- Supports imported instance names with or without ".lua".
-- ============================================================

local function findPath(root, relativePath)
	if not root then
		return nil
	end

	local current = root

	for segment in string.gmatch(relativePath, "[^/]+") do
		local child = current:FindFirstChild(segment)

		if not child then
			child = current:FindFirstChild(segment .. ".lua")
		end

		if not child then
			local baseName = segment:gsub("%.lua$", "")

			for _, candidate in ipairs(current:GetChildren()) do
				local candidateName =
					candidate.Name:gsub("%.lua$", "")

				if candidateName == baseName then
					child = candidate
					break
				end
			end
		end

		if not child then
			return nil
		end

		current = child
	end

	return current
end

-- ============================================================
-- SAFE REQUIRE
-- Only ModuleScripts can be required.
-- A nil module return is valid if require itself succeeds.
-- ============================================================

local function safeRequire(moduleScript)
	if not moduleScript then
		return false, "ModuleScript was not found."
	end

	if not moduleScript:IsA("ModuleScript") then
		local message = string.format(
			"Expected ModuleScript, got %s: %s",
			moduleScript.ClassName,
			moduleScript:GetFullName()
		)

		warnScript(message)
		return false, message
	end

	local success, result = xpcall(function()
		return require(moduleScript)
	end, traceback)

	if not success then
		warnScript(
			"Failed to require "
				.. moduleScript:GetFullName()
				.. ":\n"
				.. tostring(result)
		)

		return false, result
	end

	return true, result
end

local function safeRequirePath(root, relativePath)
	local target = findPath(root, relativePath)

	if not target then
		warnScript("Missing module: " .. relativePath)
		return false
	end

	local success = safeRequire(target)

	if success then
		log("Initialized " .. relativePath)
	end

	return success
end

-- ============================================================
-- CORESCRIPT ENTRY POINTS
--
-- Unlike AddCoreScriptLocal, this directly requires a
-- ModuleScript already installed in the historical tree.
--
-- LocalScripts cannot be started with require(). If an entry
-- point is a LocalScript, the bootstrap/execution host must
-- handle its lifecycle; this function will not clone it.
-- ============================================================

local function startCoreScript(relativePath)
	local entry = findPath(CoreScripts, relativePath)

	if not entry then
		warnScript(
			"CoreScript entry point missing: CoreScripts/"
				.. relativePath
		)

		return false
	end

	if entry:IsA("LocalScript") then
		warnScript(
			"Entry point CoreScripts/"
				.. relativePath
				.. " is a LocalScript, not a ModuleScript. "
				.. "It cannot be started with require(). "
				.. "Its lifecycle must be handled by the bootstrap."
		)

		return false
	end

	if not entry:IsA("ModuleScript") then
		warnScript(
			"Unsupported entry point type for CoreScripts/"
				.. relativePath
				.. ": "
				.. entry.ClassName
		)

		return false
	end

	local success = safeRequire(entry)

	if success then
		log("Started CoreScripts/" .. relativePath)
	end

	return success
end

local function create(className, properties, parent)
	local object = Instance.new(className)

	for property, value in pairs(properties or {}) do
		local success, err = pcall(function()
			object[property] = value
		end)

		if not success then
			warnLoading(
				"Could not set "
					.. tostring(property)
					.. " on "
					.. tostring(className)
					.. ": "
					.. tostring(err)
			)
		end
	end

	object.Parent = parent

end

local function setProperty(object, property, value)
	if not object then
		return false
	end

	return pcall(function()
		object[property] = value
	end)
end

-- ============================================================
-- PLACE NAME AND CREATOR INFORMATION
-- ============================================================

local function extractGeneratedUsername(gameName)
	if type(gameName) ~= "string" then
		return nil
	end

	local username = string.match(
		gameName,
		"^([0-9a-fA-F]+)'s Place$"
	)

	if username and #username == 32 then
		return username
	end

	return nil
end

local function getFilteredGameName(gameName, creatorName)
	if type(gameName) ~= "string" then
		return gameName
	end

	local temporaryUsername =
		extractGeneratedUsername(gameName)

	if temporaryUsername and type(creatorName) == "string" then
		return string.gsub(
			gameName,
			temporaryUsername,
			creatorName,
			1
		)
	end

	return gameName
end

local function shouldConvertXboxPlaceName()
	if platform ~= Enum.Platform.XBoxOne then
		return false
	end

	local success, value = pcall(function()
		return settings():GetFFlag(
			"ConvertMyPlaceNameInXboxApp"
		)
	end)

	return success and value == true
end

local function getGameName()
	if not gameAssetInfo then
		return ""
	end

	local name = gameAssetInfo.Name

	if type(name) ~= "string" then
		return ""
	end

	if shouldConvertXboxPlaceName() then
		return getFilteredGameName(
			name,
			getCreatorName()
		)
	end

	return name
end

local function getCreatorName()
	if not gameAssetInfo then
		return ""
	end

	local creator = gameAssetInfo.Creator

	if type(creator) ~= "table" then
		return ""
	end

	if type(creator.Name) ~= "string" then
		return ""
	end

	return creator.Name
end

local function loadPlaceInformation()
	task.spawn(function()
		local placeId = PLACE_ID
		local timeoutAt = os.clock() + PLACE_INFO_TIMEOUT

		while placeId &lt;= 0 and os.clock() &lt; timeoutAt do
			if stopping or stopped then
				return
			end

			task.wait(0.25)
			placeId = game.PlaceId
		end

		if placeId &lt;= 0 then
			return
		end

		local success, result = pcall(function()
			return MarketplaceService:GetProductInfo(placeId)
		end)

		if not success then
			warnLoading(
				"Place metadata could not be loaded: "
					.. tostring(result)
			)

			return
		end

		if type(result) ~= "table" then
			return
		end

		gameAssetInfo = result

		if stopping or stopped then
			return
		end

		if placeLabel and placeLabel.Parent then
			local name = getGameName()

			if name ~= "" then
				placeLabel.Text = name
			end
		end

		if creatorLabel and creatorLabel.Parent then
			local name = getCreatorName()

			if name ~= "" then
				if isTenFootInterface then
					local showCreator = true

					if platform == Enum.Platform.XBoxOne then
						local ok, value = pcall(function()
							return settings():GetFFlag(
								"ShowDevNameInXboxApp"
							)
						end)

						if ok then
							showCreator = value == true
						end
					end

					creatorLabel.Text =
						showCreator and name or ""

					if creatorIcon then
						creatorIcon.Visible = showCreator
					end

					if byLabel then
						byLabel.Visible = showCreator
					end
				else
					creatorLabel.Text = "By " .. name
				end
			end
		end
	end)
end

local function tileBackgroundTexture(frameToFill)
	if not frameToFill or not frameToFill.Parent then
		return
	end

	for _, child in ipairs(frameToFill:GetChildren()) do
		child:Destroy()
	end

	if backgroundImageTransparency &gt;= 1 then
		return
	end

	local absoluteSize = frameToFill.AbsoluteSize

	if absoluteSize.X &lt;= 0 or absoluteSize.Y &lt;= 0 then
		return
	end

	local columns = math.ceil(
		absoluteSize.X / BACKGROUND_TEXTURE_SIZE.X
	)

	local rows = math.ceil(
		absoluteSize.Y / BACKGROUND_TEXTURE_SIZE.Y
	)

	for x = 0, columns do
		for y = 0, rows do
			create("ImageLabel", {
				Name = "BackgroundTextureImage",

				BackgroundTransparency = 1,
				BorderSizePixel = 0,

				Image = BACKGROUND_TEXTURE,
				ImageTransparency = backgroundImageTransparency,

				Position = UDim2.new(
					0,
					x * BACKGROUND_TEXTURE_SIZE.X,
					0,
					y * BACKGROUND_TEXTURE_SIZE.Y
				),

				Size = UDim2.new(
					0,
					BACKGROUND_TEXTURE_SIZE.X,
					0,
					BACKGROUND_TEXTURE_SIZE.Y
				),

				ZIndex = 1,
			}, frameToFill)
		end
	end
end

-- ============================================================
-- FADE HELPERS
-- ============================================================

local function setObjectTransparency(object, transparency)
	if not object or not object.Parent then
		return
	end

	if object:IsA("TextLabel")
		or object:IsA("TextButton")
		or object:IsA("TextBox") then

		setProperty(object, "TextTransparency", transparency)
		setProperty(object, "TextStrokeTransparency", transparency)
		setProperty(object, "BackgroundTransparency", transparency)

	elseif object:IsA("ImageLabel")
		or object:IsA("ImageButton") then

		setProperty(object, "ImageTransparency", transparency)
		setProperty(object, "BackgroundTransparency", transparency)

	elseif object:IsA("Frame")
		or object:IsA("ScrollingFrame") then

		setProperty(object, "BackgroundTransparency", transparency)
	end
end

local function fadeAndDestroyBlackFrame(frame)
	if not frame or not frame.Parent then
		stopped = true
		return
	end

	task.spawn(function()
		local descendants = frame:GetDescendants()

		local transparency = 0
		local lastUpdate = os.clock()

		while transparency &lt; 1 do
			if not frame or not frame.Parent then
				break
			end

			local now = os.clock()
			local deltaTime = now - lastUpdate

			lastUpdate = now

			transparency = math.min(
				1,
				transparency + FADE_RATE * deltaTime
			)

			frame.BackgroundTransparency = transparency

			for _, object in ipairs(descendants) do
				setObjectTransparency(
					object,
					transparency
				)
			end

			backgroundImageTransparency = transparency

			task.wait()
		end

		if frame and frame.Parent then
			frame:Destroy()
		end

		blackFrame = nil
		closeButton = nil
		graphicsFrame = nil
		loadingImage = nil
		loadingText = nil
		uiMessageFrame = nil
		uiMessage = nil
		infoFrame = nil
		placeLabel = nil
		creatorLabel = nil
		creatorIcon = nil
		byLabel = nil
		backgroundTextureFrame = nil

		destroyedLoadingGui = true
		stopped = true

		-- Keep the ScreenGui and ErrorFrame alive so a subsequent
		-- loading or connection error can still be displayed.
		log("Loading screen removed.")
	end)
end

local function destroyLoadingElements(immediate)
	if destroyedLoadingGui then
		return
	end

	destroyedLoadingGui = true

	disconnectRendering()

	local frame = blackFrame

	if not frame then
		stopped = true
		return
	end

	if immediate then
		frame:Destroy()

		blackFrame = nil
		closeButton = nil
		graphicsFrame = nil
		loadingImage = nil
		loadingText = nil
		uiMessageFrame = nil
		uiMessage = nil
		infoFrame = nil
		placeLabel = nil
		creatorLabel = nil
		creatorIcon = nil
		byLabel = nil
		backgroundTextureFrame = nil

		stopped = true

		log("Loading screen removed immediately.")
		return
	end

	-- Reset the internal guard because fadeAndDestroyBlackFrame
	-- handles the asynchronous destruction from this point.
	destroyedLoadingGui = false

	fadeAndDestroyBlackFrame(frame)
end

-- ============================================================
-- STOP API
-- ============================================================

function LoadingScreen:Stop(immediate)
	if stopping or stopped then
		return
	end

	stopping = true

	task.spawn(function()
		-- Avoid a brief flash if the loader is stopped almost
		-- immediately after it appears.
		if not immediate then
			local elapsed = os.clock() - startedAt
			local remaining = MINIMUM_DISPLAY_TIME - elapsed

			if remaining &gt; 0 then
				task.wait(remaining)
			end
		end

		destroyLoadingElements(immediate == true)
	end)
end

	return screenGui
end

function LoadingScreen:IsStopped()
	return stopped
end

-- ============================================================
-- SPINNER, TEXT, TEXTURE AND CLOSE BUTTON ANIMATION
-- ============================================================

renderConnection = RunService.RenderStepped:Connect(function(deltaTime)
	if not blackFrame or not blackFrame.Parent then
		return
	end

	-- Rotate the original Roblox loading circle.
	if loadingImage and loadingImage.Parent then
		elapsedAnimationTime += deltaTime

		loadingImage.Rotation =
			(elapsedAnimationTime * 360
				/ SPINNER_ROTATION_TIME) % 360
	end

	-- Animate "Loading", "Loading.", "Loading..", "Loading..."
	if loadingText and loadingText.Parent then
		elapsedDotTime += deltaTime

		if elapsedDotTime &gt;= LOADING_DOT_INTERVAL then
			elapsedDotTime = elapsedDotTime
				% LOADING_DOT_INTERVAL

			loadingDotCount = (loadingDotCount + 1) % 4

			loadingText.Text =
				"Loading" .. string.rep(".", loadingDotCount)
		end
	end

	-- Recreate tiled background images when the viewport changes.
	if backgroundTextureFrame
		and backgroundTextureFrame.Parent then

		local size = backgroundTextureFrame.AbsoluteSize

		if size ~= lastBackgroundSize then
			lastBackgroundSize = size

			tileBackgroundTexture(
				backgroundTextureFrame
			)
		end
	end

	-- The classic close icon fades in after five seconds.
	if closeButton and closeButton.Parent then
		local elapsed = os.clock() - startedAt

        local sorted = mergeSort(items, function(a, b)
            return a &lt; b
        end)

    If no comparator is provided, elements are sorted in
    ascending order using the &lt; operator.

    The input table is not modified.

local function bottomupmerge(comp, source, target, left, mid, right)
	local i = left
	local j = mid

	for k = left, right do
		if i &lt; mid and (j &gt; right or not comp(source[j], source[i])) then
			target[k] = source[i]
			i = i + 1
		else
			target[k] = source[j]
			j = j + 1
		end
	end
end

local function mergesort(arr, comp)
	assert(type(arr) == "table", "MergeSort expects an array table.")

	if comp == nil then
		comp = function(a, b)
			return a &lt; b
		end
	end

	assert(type(comp) == "function", "Comparator must be a function.")

	local n = #arr
	local source = {}
	local target = {}

	-- Copy the original array so sorting never changes its contents.
	for i = 1, n do
		source[i] = arr[i]
	end

	local width = 1

	while width &lt; n do
		for left = 1, n, width * 2 do
			-- mid is the first index of the right half.
			-- right is the final included index.
			local mid = math.min(left + width, n + 1)
			local right = math.min(left + width * 2 - 1, n)

			bottomupmerge(
				comp,
				source,
				target,
				left,
				mid,
				right
			)
		end

		source, target = target, source
		width = width * 2
	end

	return source
end

local function getOrCreateRemote(name, className)
	local existing = ReplicatedStorage:FindFirstChild(name)

	if existing then
		if existing.ClassName ~= className then
			warnScript(
				"Cannot initialize "
					.. name
					.. ": expected "
					.. className
					.. ", found "
					.. existing.ClassName
			)

			return nil
		end

		return existing
	end

	local remote = Instance.new(className)
	remote.Name = name
	remote.Parent = ReplicatedStorage

	return remote
end

-- ============================================================
-- DIALOG COMPATIBILITY
-- ============================================================

-- Modern Roblox manages Dialog usage itself. The old
-- Dialog:SetPlayerIsUsing API is no longer available through
-- the documented Dialog API.
--
-- Retain the historical RemoteEvent name for compatibility,
-- but only invoke the legacy method when it actually exists.
-- Otherwise, report the limitation instead of crashing.

local playerDialogMap = {}
local warnedLegacyDialogAPI = false

local function setDialogInUse(player, dialog, value, waitTime)
	-- RemoteEvent supplies the real requesting Player.
	if typeof(player) ~= "Instance"
		or not player:IsA("Player")
		or player.Parent ~= Players then
		return
	end

	if typeof(dialog) ~= "Instance"
		or not dialog:IsA("Dialog") then
		return
	end

	-- The requested dialog must belong to the server's world.
	if not dialog:IsDescendantOf(Workspace) then
		return
	end

	if type(value) ~= "boolean" then
		return
	end

	if waitTime ~= nil then
		if type(waitTime) ~= "number"
			or waitTime ~= waitTime
			or math.abs(waitTime) == math.huge then
			return
		end
	else
		waitTime = 0
	end

	-- Avoid unbounded client-requested delays.
	waitTime = math.clamp(waitTime, 0, 5)

	local function apply()
		-- Revalidate everything after a possible delay.
		if player.Parent ~= Players then
			return
		end

		if not dialog:IsDescendantOf(Workspace) then
			return
		end

		local success, err = pcall(function()
			dialog:SetPlayerIsUsing(player, value)
		end)

		if success then
			playerDialogMap[player] = value and dialog or nil
			return
		end

		if not warnedLegacyDialogAPI then
			warnedLegacyDialogAPI = true

			warnScript(
				"The historical Dialog:SetPlayerIsUsing API "
					.. "is unavailable. Dialog usage is managed by "
					.. "the modern Roblox engine; legacy usage requests "
					.. "cannot change its read-only InUse state. "
					.. "First error: "
					.. tostring(err)
			)
		end
	end

	if waitTime &gt; 0 then
		task.delay(waitTime, apply)
	else
		apply()
	end
end

if RemoteEvent_SetDialogInUse then
	RemoteEvent_SetDialogInUse.OnServerEvent:Connect(
		setDialogInUse
	)
end

local function forceUsernameNameplates()
	local connections = setmetatable({}, { __mode = "k" })
	local function applyCharacter(player, character)
		if not character then return end
		local function bindHumanoid(humanoid)
			if not humanoid or not humanoid:IsA("Humanoid") then return end
			local function applyName()
				if humanoid.Parent and player.Parent == Players then
					pcall(function() humanoid.DisplayName = player.Name end)
				end
			end
			applyName()
			if not connections[humanoid] then
				connections[humanoid] = humanoid:GetPropertyChangedSignal("DisplayName"):Connect(function()
					if humanoid.DisplayName ~= player.Name then applyName() end
				end)
			end
		end
		local existing = character:FindFirstChildOfClass("Humanoid")
		if existing then bindHumanoid(existing) end
		character.ChildAdded:Connect(function(child)
			if child:IsA("Humanoid") then bindHumanoid(child) end
		end)
	end
	local function bindPlayer(player)
		if player.Character then applyCharacter(player, player.Character) end
		player.CharacterAdded:Connect(function(character) applyCharacter(player, character) end)
		player:GetPropertyChangedSignal("Name"):Connect(function()
			if player.Character then
				local humanoid = player.Character:FindFirstChildOfClass("Humanoid")
				if humanoid then pcall(function() humanoid.DisplayName = player.Name end) end
			end
		end)
	end
	for _, player in ipairs(Players:GetPlayers()) do bindPlayer(player) end
	Players.PlayerAdded:Connect(bindPlayer)
end
